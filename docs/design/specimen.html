// exponential backoff with jitter
export async function withRetry<T>(fn: () => Promise<T>, tries = 3) {
  for (let i = 0; i < tries; i++) {
    try { return await fn() } catch { await sleep(2 ** i * 250) }
  }
  throw new Error("retries exhausted")
}
